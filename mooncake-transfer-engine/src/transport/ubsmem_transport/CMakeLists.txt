if(USE_TENT)
  message(FATAL_ERROR "USE_UBSMEM currently requires USE_TENT=OFF")
endif()
if(UBSMEM_BACKEND STREQUAL "mock")
  set(UBSMEM_BACKEND_SOURCE ubsmem_mock.cpp)
elseif(UBSMEM_BACKEND STREQUAL "sdk")
  find_path(UBSMEM_INCLUDE_DIR ubs_mem.h)
  find_library(UBSMEM_LIBRARY ubsm_sdk)
  if(NOT UBSMEM_INCLUDE_DIR OR NOT UBSMEM_LIBRARY)
    message(FATAL_ERROR "The SDK backend requires ubs_mem.h and libubsm_sdk")
  endif()
  set(UBSMEM_BACKEND_SOURCE ubsmem_sdk.cpp)
else()
  message(FATAL_ERROR "UBSMEM_BACKEND must be mock or sdk")
endif()
add_library(ubsmem_memory STATIC ubsmem_memory.cpp ${UBSMEM_BACKEND_SOURCE})
target_include_directories(ubsmem_memory
                           PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../../../include)
target_link_libraries(ubsmem_memory PUBLIC pthread)
if(UBSMEM_BACKEND STREQUAL "sdk")
  target_include_directories(ubsmem_memory PRIVATE ${UBSMEM_INCLUDE_DIR})
  target_link_libraries(ubsmem_memory PUBLIC ${UBSMEM_LIBRARY})
endif()
add_library(ubsmem_transport OBJECT ubsmem_transport.cpp)
target_link_libraries(ubsmem_transport PRIVATE ubsmem_memory glog::glog
                                               JsonCpp::JsonCpp)
